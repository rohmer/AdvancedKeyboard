cmake_minimum_required(VERSION 3.24)

# ESP-IDF evaluates a component before ordinary host-project setup.
if(COMMAND idf_component_register)
    include("${CMAKE_CURRENT_LIST_DIR}/cmake/AdvancedKeyboardEspIdf.cmake")
    return()
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdvancedKeyboard VERSION 0.1.0 LANGUAGES C CXX)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/cmake/AdvancedKeyboardOptions.cmake")
include("${CMAKE_CURRENT_LIST_DIR}/cmake/AdvancedKeyboardSources.cmake")

if(ADVKBD_BUILD_WIDGET OR ADVKBD_BUILD_PREDICTION)
    get_property(_advkbd_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
    if(NOT "CXX" IN_LIST _advkbd_languages)
        enable_language(CXX)
    endif()
endif()

if(ADVKBD_BUILD_PREDICTION)
    advkbd_prediction_sources(_advkbd_prediction_sources)
    add_library(advanced_keyboard_prediction_objects OBJECT ${_advkbd_prediction_sources})
    target_compile_features(advanced_keyboard_prediction_objects PRIVATE cxx_std_17)
    target_include_directories(advanced_keyboard_prediction_objects PRIVATE
        "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_compile_definitions(advanced_keyboard_prediction_objects PRIVATE
        ESPOS_KEYBOARD_HEADLESS_PREDICTION=1
        LVSERIAL_USE_ADVANCED_KEYBOARD_PREDICTION=1 LVSERIAL_USE_ADVANCED_KEYBOARD_BUILTIN_TRIE=0)
    add_library(advanced_keyboard_prediction STATIC $<TARGET_OBJECTS:advanced_keyboard_prediction_objects>)
    add_library(AdvancedKeyboard::Prediction ALIAS advanced_keyboard_prediction)
    target_compile_features(advanced_keyboard_prediction PUBLIC cxx_std_17)
    target_include_directories(advanced_keyboard_prediction PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}/include")
    if(MSVC)
        target_compile_options(advanced_keyboard_prediction_objects PRIVATE /utf-8 /EHsc)
    else()
        target_compile_options(advanced_keyboard_prediction_objects PRIVATE -fexceptions)
    endif()
endif()

if(ADVKBD_BUILD_WIDGET)
    get_property(_advkbd_languages GLOBAL PROPERTY ENABLED_LANGUAGES)
    if(NOT "C" IN_LIST _advkbd_languages)
        enable_language(C)
    endif()
    if(ADVKBD_LVGL_SOURCE_DIR AND NOT TARGET lvgl AND NOT TARGET lvgl::lvgl)
        if(NOT EXISTS "${ADVKBD_LVGL_SOURCE_DIR}/CMakeLists.txt")
            message(FATAL_ERROR "ADVKBD_LVGL_SOURCE_DIR must name an existing LVGL 9.6 source checkout")
        endif()
        # Local-checkout convenience must not turn on upstream demo/example
        # targets or network dependency fetching. Explicit caller choices win.
        set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "Fetch LVGL optional dependencies")
        set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "Build LVGL demos")
        set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "Build LVGL examples")
        set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "Build LVGL ThorVG backend")
        add_subdirectory("${ADVKBD_LVGL_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl" EXCLUDE_FROM_ALL)
    endif()
    set(_advkbd_lvgl "${ADVKBD_LVGL_TARGET}")
    if(NOT _advkbd_lvgl)
        if(TARGET lvgl::lvgl)
            set(_advkbd_lvgl lvgl::lvgl)
        elseif(TARGET lvgl)
            set(_advkbd_lvgl lvgl)
        endif()
    endif()
    if(NOT _advkbd_lvgl OR NOT TARGET "${_advkbd_lvgl}")
        message(FATAL_ERROR "Add LVGL 9.6 before AdvancedKeyboard, set ADVKBD_LVGL_TARGET, or provide ADVKBD_LVGL_SOURCE_DIR. For native tools/tests without LVGL set ADVKBD_BUILD_WIDGET=OFF.")
    endif()
    get_target_property(_advkbd_actual_lvgl "${_advkbd_lvgl}" ALIASED_TARGET)
    if(NOT _advkbd_actual_lvgl)
        set(_advkbd_actual_lvgl "${_advkbd_lvgl}")
    endif()
    advkbd_widget_sources(_advkbd_widget_sources)
    advkbd_registry_paths(_advkbd_registry_sources _advkbd_registry_cpp _advkbd_registry_report)
    list(APPEND _advkbd_widget_sources ${_advkbd_registry_sources})
    if(_advkbd_registry_cpp)
        advkbd_add_registry_rule("${_advkbd_registry_cpp}" "${_advkbd_registry_report}")
    endif()
    advkbd_feature_definitions(_advkbd_definitions)
    if(ADVKBD_EMBED_IN_LVGL)
        get_target_property(_advkbd_imported "${_advkbd_actual_lvgl}" IMPORTED)
        get_target_property(_advkbd_type "${_advkbd_actual_lvgl}" TYPE)
        if(_advkbd_imported OR _advkbd_type STREQUAL "INTERFACE_LIBRARY")
            message(FATAL_ERROR "Embedding requires a buildable LVGL library target; imported/interface targets support the default separate-library mode")
        endif()
        target_sources("${_advkbd_actual_lvgl}" PRIVATE ${_advkbd_widget_sources})
        target_compile_features("${_advkbd_actual_lvgl}" PRIVATE c_std_99 cxx_std_17)
        target_include_directories("${_advkbd_actual_lvgl}" PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
            PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions("${_advkbd_actual_lvgl}" PUBLIC ${_advkbd_definitions})
        if(ADVKBD_BUILD_PREDICTION)
            target_sources("${_advkbd_actual_lvgl}" PRIVATE $<TARGET_OBJECTS:advanced_keyboard_prediction_objects>)
            add_dependencies("${_advkbd_actual_lvgl}" advanced_keyboard_prediction_objects)
            set_target_properties(advanced_keyboard_prediction PROPERTIES EXCLUDE_FROM_ALL TRUE)
            if(_advkbd_type STREQUAL "SHARED_LIBRARY" OR _advkbd_type STREQUAL "MODULE_LIBRARY")
                set_target_properties(advanced_keyboard_prediction_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
            endif()
        endif()
        if(MSVC)
            target_compile_options("${_advkbd_actual_lvgl}" PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/EHsc> /utf-8)
        else()
            target_compile_options("${_advkbd_actual_lvgl}" PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fexceptions>)
        endif()
        add_library(advanced_keyboard INTERFACE)
        target_link_libraries(advanced_keyboard INTERFACE "${_advkbd_lvgl}")
    else()
        add_library(advanced_keyboard STATIC ${_advkbd_widget_sources})
        target_compile_features(advanced_keyboard PUBLIC c_std_99 cxx_std_17)
        target_include_directories(advanced_keyboard PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
            PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_SOURCE_DIR}")
        target_compile_definitions(advanced_keyboard PUBLIC ${_advkbd_definitions})
        target_link_libraries(advanced_keyboard PUBLIC "${_advkbd_lvgl}")
        if(ADVKBD_BUILD_PREDICTION)
            target_link_libraries(advanced_keyboard PRIVATE advanced_keyboard_prediction)
        endif()
        if(MSVC)
            target_compile_options(advanced_keyboard PRIVATE /utf-8 $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
        else()
            target_compile_options(advanced_keyboard PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-fexceptions>)
        endif()
    endif()
    add_library(AdvancedKeyboard::AdvancedKeyboard ALIAS advanced_keyboard)
elseif(ADVKBD_BUNDLED_MODELS AND NOT ADVKBD_BUNDLED_MODELS STREQUAL "none")
    message(FATAL_ERROR "The bundled-model widget registry requires ADVKBD_BUILD_WIDGET=ON")
endif()

if(ADVKBD_BUILD_TOOLS)
    add_subdirectory(tools)
endif()
if(ADVKBD_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()
if(ADVKBD_BUILD_EXAMPLE)
    add_subdirectory(examples/Basic)
endif()
