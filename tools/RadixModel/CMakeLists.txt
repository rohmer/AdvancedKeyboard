cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(RadixModelConverter LANGUAGES CXX)
endif()
if(NOT TARGET AdvancedKeyboard::ModelCodec)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../ModelCodec" "${CMAKE_CURRENT_BINARY_DIR}/model-codec")
endif()
add_executable(UICreatorRadixConvert ModelConvert.cpp)
target_compile_features(UICreatorRadixConvert PRIVATE cxx_std_17)
target_link_libraries(UICreatorRadixConvert PRIVATE AdvancedKeyboard::ModelCodec)
if(MSVC)
    target_compile_options(UICreatorRadixConvert PRIVATE /utf-8 /EHsc)
endif()
