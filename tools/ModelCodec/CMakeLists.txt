cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdvancedKeyboardModelCodec LANGUAGES CXX)
endif()
get_filename_component(advanced_keyboard_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(AdvancedKeyboardModelCodec STATIC RadixModelBuilder.cpp)
add_library(AdvancedKeyboard::ModelCodec ALIAS AdvancedKeyboardModelCodec)
target_compile_features(AdvancedKeyboardModelCodec PUBLIC cxx_std_17)
target_include_directories(AdvancedKeyboardModelCodec PUBLIC "${CMAKE_CURRENT_LIST_DIR}"
    PRIVATE "${advanced_keyboard_root}/src" "${advanced_keyboard_root}/include")
if(TARGET AdvancedKeyboard::Prediction)
    target_link_libraries(AdvancedKeyboardModelCodec PUBLIC AdvancedKeyboard::Prediction)
else()
    # Direct tool-only builds require just the model reader, not LVGL or board code.
    target_sources(AdvancedKeyboardModelCodec PRIVATE "${advanced_keyboard_root}/src/PredictionModel.cpp")
endif()
if(MSVC)
    target_compile_options(AdvancedKeyboardModelCodec PRIVATE /utf-8 /EHsc)
endif()
