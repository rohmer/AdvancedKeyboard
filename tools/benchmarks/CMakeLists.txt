cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(AdvancedKeyboardBenchmarks LANGUAGES CXX)
endif()
get_filename_component(advanced_keyboard_root "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(KEYBOARD_CORE "${advanced_keyboard_root}/src" CACHE PATH "Standalone or explicitly frozen prediction sources")
set(KEYBOARD_PUBLIC_INCLUDE "${advanced_keyboard_root}/include" CACHE PATH "Matching public keyboard headers")
set(KEYBOARD_CODEC "${advanced_keyboard_root}/tools/ModelCodec" CACHE PATH "Matching offline radix encoder sources")
if(NOT EXISTS "${KEYBOARD_CORE}/PredictionModel.h")
    message(FATAL_ERROR "KEYBOARD_CORE must contain PredictionModel.h from this package or a frozen snapshot")
endif()
file(READ "${KEYBOARD_CORE}/PredictionModel.h" model_header)
set(has_scores 0)
set(has_physical_nodes 0)
if(model_header MATCHES "ConditionalLogProbability")
    set(has_scores 1)
endif()
if(model_header MATCHES "PhysicalNodeCount")
    set(has_physical_nodes 1)
endif()
foreach(tool PredictionProbe RuntimeProfile)
    add_executable(${tool} ${tool}.cpp "${KEYBOARD_CORE}/PredictionModel.cpp"
        "${KEYBOARD_CORE}/PredictionPrefix.cpp" "${KEYBOARD_CORE}/Utf8Fold.cpp")
    target_compile_features(${tool} PRIVATE cxx_std_17)
    target_compile_definitions(${tool} PRIVATE ESPOS_KEYBOARD_HEADLESS_PREDICTION=1
        LVSERIAL_USE_ADVANCED_KEYBOARD_PREDICTION=1 BENCH_HAS_SCORES=${has_scores}
        BENCH_HAS_PHYSICAL_NODES=${has_physical_nodes} BENCH_MODEL_TRACE=0)
    target_include_directories(${tool} PRIVATE "${KEYBOARD_CORE}" "${KEYBOARD_PUBLIC_INCLUDE}")
    if(MSVC)
        target_compile_options(${tool} PRIVATE /utf-8 /EHsc)
    endif()
endforeach()
if(EXISTS "${KEYBOARD_CODEC}/RadixModelBuilder.cpp" AND NOT TARGET RadixConvert)
    add_executable(RadixConvert RadixConvert.cpp "${KEYBOARD_CODEC}/RadixModelBuilder.cpp" "${KEYBOARD_CORE}/PredictionModel.cpp")
    target_compile_features(RadixConvert PRIVATE cxx_std_17)
    target_include_directories(RadixConvert PRIVATE "${KEYBOARD_CORE}" "${KEYBOARD_PUBLIC_INCLUDE}" "${KEYBOARD_CODEC}")
    if(MSVC)
        target_compile_options(RadixConvert PRIVATE /utf-8 /EHsc)
    endif()
endif()
